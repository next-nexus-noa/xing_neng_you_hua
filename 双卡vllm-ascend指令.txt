 --name vllm-ascend-adaptive \

======================3.添加环境变量命令======================
###3.1服务端###
export VLLM_USE_MODELSCOPE=True
export PYTORCH_NPU_ALLOC_CONF=max_split_size_mb:256 
export EXTRA_SERVER_ARGS="${EXTRA_SERVER_ARGS} --no-enable-prefix-caching"
export MICROBATCH_ENV_NAME=VLLM_ASCEND_PP_MICROBATCH
export PLAN=gradient_plan_test.csv
export RESULT_ROOT=gradient_results
export ASCEND_DEVICES=0,1
export TRACE_ENABLED=1
export TRACE_FILE=/workspace/vllm_ascend_pp_trace.jsonl
export EXTRA_SERVER_ARGS='--additional-config {"sarathi_scheduler_config":{"enabled":true,"chunk_size":2048}}'

###3.2客户端###
export VLLM_USE_MODELSCOPE=True
export PYTORCH_NPU_ALLOC_CONF=max_split_size_mb:256
export PLAN=gradient_plan_test.csv
export RESULT_ROOT=gradient_results
export TRACE_FILE=/workspace/vllm_ascend_pp_trace.jsonl

export VLLM_ASCEND_PP_TRACE=1 #选加，用于开启记录阶段时间功能
export VLLM_ASCEND_PP_TRACE_FILE=/workspace/vllm_ascend_pp_trace.jsonl #选加，设置阶段时间文件保存位置

======================6.tmux离线进行实验======================
tmux new -s wzj_experiment #创建一个tmux会话
ctrl+B C #创建一个新的窗口（同一个tmux会话下）
ctrl+B 0/1/... #切换到对应窗口
ctrl+B w #展示窗口列表
ctrl+B D #退出tmux会话窗口（此时可以断开连接，而实验依然在运行）
tmux attach -t wzj_experiment #回到之前的会话窗口
tmux kill-session -t wzj_experiment #删除会话
